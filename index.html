    function render(t) {
  const local = t - scene.start;
  scene.items.forEach((it) => {
    const p = clamp((local - it.delay) / it.dur);
    const e = ease[it.curve](p);
    it.el.style.transform = `translateY(${(1 - e) * 4}u)`;
    it.el.style.opacity = p;
  });
}

        Como funciona

        O eixo X é o tempo e o Y é o valor. Para cada quadro, o código encontra o t da curva com o método de Newton e devolve o Y. Valores acima de 1 são o "passar do ponto".
